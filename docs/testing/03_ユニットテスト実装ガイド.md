## 3. Next.js 14 App Router対応テスト

### 3.1 Route Handlers（API Routes）のテスト

Next.js 14のRoute Handlersは、従来のAPI Routesと異なり、`NextRequest`と`NextResponse`を使用します。

#### 実践例: キャリア初期データ取得APIのテスト

プロジェクト内の既存テスト `/home/user/skill-report-web/src/app/api/career/init/route.test.ts:1` を参考にした実装：

```typescript
// src/app/api/career/init/route.test.ts
import { NextRequest } from 'next/server'

/**
 * テストヘルパー: NextRequestのモック作成
 */
function createMockRequest(headers: Record<string, string> = {}) {
  const defaultHeaders: Record<string, string> = {
    'x-user-id': 'emp_001',
    ...headers
  }

  return {
    headers: {
      get: (name: string) => defaultHeaders[name.toLowerCase()] || null
    }
  } as unknown as NextRequest
}

/**
 * 動的インポートでGET関数を取得
 * Next.js App RouterのRoute Handlersは動的インポートでテストする
 */
async function importGETFunction() {
  const module = await import('./route')
  return module.GET
}

describe('API-700: キャリア初期データ取得API', () => {
  describe('GET /api/career/init', () => {
    describe('正常系', () => {
      test('キャリア初期データを正常に取得できること', async () => {
        const GET = await importGETFunction()
        const request = createMockRequest()

        const response = await GET(request)
        const responseData = await response.json()

        expect(response.status).toBe(200)
        expect(responseData.success).toBe(true)
        expect(responseData.data).toHaveProperty('career_goal')
        expect(responseData.data).toHaveProperty('skill_categories')
        expect(responseData.data).toHaveProperty('positions')
      })

      test('ユーザーIDがヘッダーにない場合、デフォルトユーザーIDを使用すること', async () => {
        const GET = await importGETFunction()
        const request = createMockRequest({ 'x-user-id': '' })

        const response = await GET(request)
        const responseData = await response.json()

        expect(response.status).toBe(200)
        expect(responseData.success).toBe(true)
      })
    })

    describe('レスポンス形式テスト', () => {
      test('レスポンスが正しい形式であること', async () => {
        const GET = await importGETFunction()
        const request = createMockRequest()

        const response = await GET(request)
        const responseData = await response.json()

        expect(responseData).toHaveProperty('success')
        expect(responseData).toHaveProperty('data')
        expect(responseData).toHaveProperty('timestamp')
        expect(typeof responseData.success).toBe('boolean')
        expect(typeof responseData.timestamp).toBe('string')
      })
    })

    describe('パフォーマンステスト', () => {
      test('レスポンス時間が1秒以内であること', async () => {
        const GET = await importGETFunction()
        const request = createMockRequest()

        const startTime = Date.now()
        const response = await GET(request)
        const endTime = Date.now()
        const executionTime = endTime - startTime

        expect(response.status).toBe(200)
        expect(executionTime).toBeLessThan(1000) // 1秒以内
      })

      test('複数回の連続リクエストが正常に処理されること', async () => {
        const GET = await importGETFunction()
        const request = createMockRequest()

        const promises = Array.from({ length: 5 }, () => GET(request))
        const responses = await Promise.all(promises)

        responses.forEach(response => {
          expect(response.status).toBe(200)
        })
      })
    })
  })
})
```

#### Route Handlersテストのベストプラクティス

1. **動的インポート**: Route Handlersは動的インポートでテスト
2. **NextRequestモック**: `createMockRequest`ヘルパーを使用
3. **レスポンス形式検証**: `response.json()`で内容確認
4. **パフォーマンス**: 実行時間も検証

## 4. Prismaモック戦略

### 4.1 Prismaクライアントのモック

Prismaを使用したテストでは、実際のDBに接続せず、モックを使用します。

#### 方法1: jest-mock-extended を使用（推奨）

```bash
npm install -D jest-mock-extended
```

```typescript
// src/__mocks__/prisma.ts
import { PrismaClient } from '@prisma/client'
import { mockDeep, mockReset, DeepMockProxy } from 'jest-mock-extended'

jest.mock('@/database/prisma/client', () => ({
  __esModule: true,
  prisma: mockDeep<PrismaClient>(),
}))

beforeEach(() => {
  mockReset(prismaMock)
})

export const prismaMock = prisma as unknown as DeepMockProxy<PrismaClient>
```

#### 使用例: スキル取得APIのテスト

```typescript
// src/app/api/skills/[userId]/route.test.ts
import { prismaMock } from '@/__mocks__/prisma'

describe('GET /api/skills/[userId]', () => {
  test('ユーザーのスキル一覧を取得', async () => {
    const mockSkills = [
      {
        skill_id: 'sk_001',
        skill_name: 'JavaScript',
        skill_level: '○',
        experience_years: 3,
      },
    ]

    // Prismaのモック設定
    prismaMock.tRN_UserSkill.findMany.mockResolvedValue(mockSkills as any)

    const GET = await import('./route').then(m => m.GET)
    const request = createMockRequest({ 'x-user-id': 'emp_001' })

    const response = await GET(request, { params: { userId: 'emp_001' } })
    const data = await response.json()

    expect(data.success).toBe(true)
    expect(data.data).toHaveLength(1)
    expect(data.data[0].skill_name).toBe('JavaScript')
    expect(prismaMock.tRN_UserSkill.findMany).toHaveBeenCalledWith({
      where: { employee_id: 'emp_001' },
    })
  })

  test('DBエラー時のエラーハンドリング', async () => {
    prismaMock.tRN_UserSkill.findMany.mockRejectedValue(
      new Error('Database connection failed')
    )

    const GET = await import('./route').then(m => m.GET)
    const request = createMockRequest()

    const response = await GET(request, { params: { userId: 'emp_001' } })
    const data = await response.json()

    expect(response.status).toBe(500)
    expect(data.success).toBe(false)
    expect(data.error).toContain('Database')
  })
})
```

#### src/__mocks__/handlers.ts（MSW v2 API）
```typescript
import { http, HttpResponse } from 'msw'

export const handlers = [
  // 認証API
  http.post('/api/auth/login', async ({ request }) => {
    const { email, password } = await request.json()

    if (email === 'user@example.com' && password === 'password') {
      return HttpResponse.json(
        {
          success: true,
          token: 'mock-jwt-token',
          user: {
            id: 1,
            name: 'テストユーザー',
            email: 'user@example.com',
            role: 'user'
          }
        },
        { status: 200 }
      )
    }

    return HttpResponse.json(
      {
        success: false,
        error: '認証に失敗しました'
      },
      { status: 401 }
    )
  }),

  http.post('/api/auth/logout', () => {
    return HttpResponse.json({ success: true }, { status: 200 })
  }),

  // スキル管理API
  http.get('/api/skills', () => {
    return HttpResponse.json([
      {
        id: 1,
        name: 'JavaScript',
        category: 'プログラミング言語',
        level: '○',
        experience: '3年'
      },
      {
        id: 2,
        name: 'TypeScript',
        category: 'プログラミング言語',
        level: '◎',
        experience: '2年'
      }
    ])
  }),

  http.post('/api/skills', async ({ request }) => {
    const skill = await request.json()
    return HttpResponse.json(
      {
        id: Date.now(),
        ...skill,
        createdAt: new Date().toISOString()
      },
      { status: 201 }
    )
  }),

  http.put('/api/skills/:id', async ({ request, params }) => {
    const { id } = params
    const skill = await request.json()
    return HttpResponse.json({
      id: Number(id),
      ...skill,
      updatedAt: new Date().toISOString()
    })
  }),

  http.delete('/api/skills/:id', () => {
    return new HttpResponse(null, { status: 204 })
  }),
]
```

### 5.2 MSW v2 の主な変更点

| v1 | v2 |
|----|----|
| `rest.get()` | `http.get()` |
| `rest.post()` | `http.post()` |
| `res(ctx.json(...))` | `HttpResponse.json(...)` |
| `res(ctx.status(200))` | `HttpResponse.json(..., { status: 200 })` |
